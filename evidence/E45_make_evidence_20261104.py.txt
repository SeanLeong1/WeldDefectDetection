# A fixed, valid candidate patch from the training population.
train=read(DATA/'datasets/train/index.json')
row=next(r for r in train if r['scene_id']=='train-000-open_intrinsic-c4-r0')
asset_hash(row['raw_path']);asset_hash(row['geometry_path'])
raw=np.load(row['raw_path']);img=base.normalize(raw,norm)
with np.load(row['geometry_path']) as z:mask=z['mask']
candidates=[(int(base.tile(mask,y,x).sum()),y,x) for y in (32,64,96,128) for x in range(16,5024,32)]
_,cy,cx=max(candidates,key=lambda q:q[0])
patch=base.tile(img,cy,cx);target=base.tile(mask,cy,cx)
x=torch.from_numpy(patch.copy())[None,None]
stage_names=['enc1','enc2','bottleneck','dec2','dec1']
acts={};weights={};probs={};layer_stats=[]
for epoch,path in paths.items():
    model=TinyDefectSegNet(in_channels=1,base_channels=24)
    model.load_state_dict(torch.load(path,map_location='cpu',weights_only=False)['state_dict']);model.eval()
    captured={};handles=[]
    for name in stage_names:
        handles.append(getattr(model,name).register_forward_hook(lambda m,i,o,n=name:captured.__setitem__(n,o.detach().cpu().numpy()[0].copy())))
    leaf=[]
    if epoch==20:
        for name,module in model.named_modules():
            if isinstance(module,(torch.nn.Conv2d,torch.nn.ConvTranspose2d,torch.nn.MaxPool2d,torch.nn.BatchNorm2d)):
                def hook(m,i,o,n=name):
                    leaf.append(dict(name=n,type=type(m).__name__,input=list(i[0].shape),output=list(o.shape),parameters=sum(p.numel() for p in m.parameters()),kernel=list(m.kernel_size) if hasattr(m,'kernel_size') and isinstance(m.kernel_size,tuple) else getattr(m,'kernel_size',None)))
                handles.append(module.register_forward_hook(hook))
    with torch.inference_mode():probs[epoch]=torch.sigmoid(model(x)).numpy()[0,0]
    acts[epoch]=captured;weights[epoch]=model.enc1.block[0].weight.detach().numpy()[:,0].copy()
    for h in handles:h.remove()
    if epoch==20:
        layer_stats=leaf
        parameters=sum(p.numel() for p in model.parameters())
        block_params={name:sum(p.numel() for p in getattr(model,name).parameters()) for name in ['enc1','enc2','bottleneck','up2','dec2','up1','dec1','head']}
assert parameters==262777
selected_channels={n:np.argsort(acts[20][n].var(axis=(1,2)))[-4:][::-1].tolist() for n in stage_names}

fig,axes=plt.subplots(1,7,figsize=(17,3.3),layout='constrained')
arrays=[patch]+[acts[20][n].mean(0) for n in stage_names]+[probs[20]]
titles=['Input TOPO\n1 × 64 × 64']+[f'{n}\n'+ ' × '.join(map(str,acts[20][n].shape)) for n in stage_names]+['Sigmoid score\n1 × 64 × 64']
for i,(ax,a,title) in enumerate(zip(axes,arrays,titles)):
    ax.imshow(a,origin='lower',cmap='gray' if i==0 else 'magma',vmin=0,vmax=1 if i in (0,6) else max(float(a.max()),1e-8));ax.set_title(title,fontsize=11);ax.set_xticks([]);ax.set_yticks([])
    if i in (0,6):ax.contour(target,levels=[.5],colors=['#00c878'],linewidths=.8)

# Validate and illustrate actual failure masks from full-ring saved predictions.
index=read(DATA/'datasets/test/index.json');lookup={r['scene_id']:i for i,r in enumerate(index)}
scores={k:np.load(DATA/f'confirmation/{fname}.npy',mmap_mode='r') for k,fname in [('median','mixed_96_median'),('cnn',f'mixed_96_cnn_{seed}')]}
results=read(DATA/'results.json')
selected={k:next(v for v in results.values() if v['kind']==k and v['arm']=='mixed_96' and (k=='median' or v['seed']==seed)) for k in ['median','cnn']}
failures={};poster_panels=[]
fig,axes=plt.subplots(2,3,figsize=(11,7.6),layout='constrained')
for r,sid in enumerate(['test-004-open_intrinsic-c64-r0','test-006-open_intrinsic-c64-r0']):
    row=index[lookup[sid]];asset_hash(row['raw_path']);asset_hash(row['geometry_path'])
    image=base.normalize(np.load(row['raw_path']),norm)[16:144]
    with np.load(row['geometry_path']) as z:truth=z['mask'][16:144]
    tb=metrics.regions(truth)[1];b=tb[0];yc=int(round((b[1]+b[3])/2));xc=int(round((b[0]+b[2])/2))%5024
    y0=min(max(0,yc-16),96);ys=slice(y0,y0+32);xs=(np.arange(32)+xc-16)%5024
    predictions={};summ={}
    for k in ['median','cnn']:
        op=spec[k]['selection']['best'];pred,boxes=metrics.binary(scores[k][lookup[sid]],op['threshold'],op['min_area'],op['closing'])
        rec=next(v for v in selected[k]['rows'] if v['scene_id']==sid);calc=metrics.match(boxes,tb,5024)
        for key in ['tp','fp','fn']:assert calc[key]==rec[key]
        iou=max([metrics.circular_iou(a,b,5024) for a in boxes for b in tb] or [0.])
        predictions[k]=pred;summ[k]={**{key:int(rec[key]) for key in ['tp','fp','fn','predicted_pixels']},'best_box_iou':float(iou)}
    for col,k in enumerate(['input','median','cnn']):
        ax=axes[r,col];a=image if k=='input' else predictions[k]
        ax.imshow(a[ys][:,xs],origin='lower',cmap='gray',vmin=0,vmax=1);ax.contour(truth[ys][:,xs],levels=[.5],colors=['#00c878'],linestyles='--',linewidths=1.3)
        ax.set_xticks([]);ax.set_yticks([])
        ax.set_title('Input TOPO' if k=='input' else f"{k.upper()} mask · box IoU {summ[k]['best_box_iou']:.3f}",fontsize=11)
        if col==0:ax.set_ylabel(f"{row['background_id']}\nD = {row['diameter_mm']:.3f} mm\n{int(truth.sum())} target pixels",fontsize=11)
    failures[sid]=dict(diameter_mm=row['diameter_mm'],truth_pixels=int(truth.sum()),methods=summ)
    poster_panels.append(([image[ys][:,xs],predictions['median'][ys][:,xs],predictions['cnn'][ys][:,xs]],truth[ys][:,xs]))
